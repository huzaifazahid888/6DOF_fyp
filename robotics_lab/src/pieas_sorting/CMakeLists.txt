cmake_minimum_required(VERSION 3.16)
project(pieas_sorting)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz-sim REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-plugin REQUIRED)
find_package(gz_transport_vendor REQUIRED)
find_package(gz-transport REQUIRED)
add_library(pieas_factory_system SHARED src/factory_system.cpp)
target_compile_features(pieas_factory_system PUBLIC cxx_std_17)
target_compile_options(pieas_factory_system PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(pieas_factory_system gz-sim::gz-sim gz-plugin::register gz-transport::gz-transport)
install(TARGETS pieas_factory_system LIBRARY DESTINATION lib)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/perception scripts/sort_batch scripts/check_cell DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch docs DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_environment_hooks("${CMAKE_CURRENT_SOURCE_DIR}/hooks/pieas_sorting.dsv.in")
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(factory_core test/test_core.py TIMEOUT 180)
endif()
ament_package()
